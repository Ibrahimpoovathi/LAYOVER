   358→  async function makePostcard() {
   359→    if (!plan) return;
   360→    setPostcardLoading(true);
   361→    setPostcardUrl(null);
   362→    setPostcardFallback(false);
   363→    try {
   364→      // Postcards are sent from where you're headed: in WHEELS the stop is the
   365→      // origin, so illustrate the destination town instead.
   366→      const postcardCity =
   367→        planMode === "road" ? (plan.destination?.city ?? plan.airport.city) : plan.airport.city;
   368→      const res = await fetch("/api/ai/postcard", {
   369→        method: "POST",
   370→        headers: { "Content-Type": "application/json" },
   371→        body: JSON.stringify({
   372→          city: postcardCity,
   373→          country: plan.airport.country ?? "",
   374→          vibe: plan.vibe,
   375→          mode: planMode,
   376→          vehicle: plan.vehicle,
   377→        }),
   378→      });
   379→      const data = await res.json();
   380→      if (!data.ok || !data.id) {
   381→        setPostcardFallback(true);
   382→        return;
   383→      }
   384→      // Poll the press with short requests — no proxy can sever those.
   385→      // Cached + showcase presses finish instantly (their POST already says
   386→      // status:"done"), so skip the initial wait for those.
   387→      const instant = data.status === "done";
   388→      const deadline = Date.now() + 150_000;
   389→      let polls = 0;
   390→      let pollErrors = 0;
   391→      while (Date.now() < deadline) {
   392→        await new Promise((r) => setTimeout(r, polls === 0 ? (instant ? 0 : 1_500) : 2_500));
   393→        polls++;
   394→        try {
   395→          const pr = await fetch(`/api/ai/postcard?id=${encodeURIComponent(data.id)}`);
   396→          if (!pr.ok) throw new Error(`poll ${pr.status}`);
   397→          const pj = await pr.json();
   398→          if (pj.status === "done" && pj.imageDataUrl) {
   399→            setPostcardUrl(pj.imageDataUrl);
   400→            return;
   401→          }
   402→          if (pj.status === "jammed") {
   403→            setPostcardFallback(true);
   404→            return;
   405→          }
   406→        } catch {
   407→          pollErrors++;
   408→          if (pollErrors >= 4) {
   409→            setPostcardFallback(true);
   410→            return;
   411→          }
   412→        }
   413→      }
   414→      setPostcardFallback(true); // press overran its deadline
   415→    } catch {
   416→      setPostcardFallback(true);
   417→    } finally {
   418→      setPostcardLoading(false);
   419→    }
   420→  }
