   179→              country: airport!.country,
   180→              durationMin: duration,

   183→              forceDemo: demoMode,

   193→              mode: currentMode,
   194→              forceDemo: demoMode,

   197→      const res = await fetch("/api/ai/plan", {
   198→        method: "POST",
   199→        headers: { "Content-Type": "application/json" },
   200→        body: JSON.stringify(body),
   201→      });
   202→      const data = await res.json();

   205→      setPlan(data.plan);
   206→      setSource(data.source ?? "ai");

   210→          title: currentMode === "road" ? "Route control unreachable" : "Gate control unreachable",
   211→          description: "Dispatched a backup adventure from the demo locker instead.",
   212→        });
   213→      }
   214→      setTimeout(() => resultRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }), 120);
   215→
   216→      // archive trip (fire & forget)
   217→      saveTrip(data.plan, data.source);
   218→    } catch {
   219→      // total network failure → local demo plan so the show goes on
   220→      let local: Plan;
   221→      if (currentMode === "road" && origin && destination) {
   222→        local = genericWheelsPlan(
   223→          origin.iata,
   224→          origin.city,
   225→          origin.country,
   226→          destination.iata,
   227→          destination.city,
   228→          vehicle,
   229→          vibe
   230→        );

   232→        local = genericRoadPlan(origin.iata, origin.city, origin.country, vibe);
   233→      } else if (airport) {
   234→        local = genericDemoPlan(airport.iata, airport.city, airport.country, vibe);
   235→      } else {
   236→        return;
   237→      }
   238→      setPlan(local);
   239→      setSource("demo");
   240→      setStep("ready");
   241→      toast({
   242→        title: "Running on internal power",
   243→        description: "Network is down — the adventure continues regardless.",
   244→      });
   245→      setTimeout(() => resultRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }), 120);
   246→    }

   249→  async function saveTrip(p: Plan, src: string) {
   250→    try {
   251→      const res = await fetch("/api/trips", {
   252→        method: "POST",
   253→        headers: { "Content-Type": "application/json" },
   254→        body: JSON.stringify({ nickname, plan: p, source: src }),
   255→      });
   256→      const data = await res.json();
   257→      if (data.ok && data.trip?.missions) {
   258→        setMissionIds(data.trip.missions.map((m: { id: string }) => m.id));
   259→      }
   260→      loadWall();
   261→    } catch {
   262→      /* archiving is best-effort */
   263→    }
   264→  }

   266→  function toggleMission(i: number) {
   267→    const next = { ...done, [i]: !done[i] };
   268→    setDone(next);
   269→    const missionId = missionIds[i];
   270→    if (missionId) {
   271→      fetch(`/api/missions/${missionId}`, {
   272→        method: "PATCH",
   273→        headers: { "Content-Type": "application/json" },
   274→        body: JSON.stringify({ done: next[i] }),
   275→      }).catch(() => {});
   276→    }
   277→  }

   279→  async function playAnnouncement() {
   280→    if (!plan) return;
   281→    setAnnouncing(true);
   282→    try {
   283→      const text =
   284→        planMode === "road"
   285→          ? `Attention please. The ${plan.airport.city} to ${plan.destination?.city ?? "parts unknown"} ${plan.vehicle === "train" ? "express" : "service"} is now boarding at platform infinity. ${plan.missions.length} journey stages, ${fmtDuration(plan.totalDurationMin)} on the road. Chai is not optional. Wheels control, out.`
   286→          : `Attention travellers. The ${plan.airport.city} ${passMeta.announceKind} is now boarding at the gate. ${plan.missions.length} missions, ${fmtDuration(plan.totalDurationMin)} window. Keep your sense of wonder handy. Gate control, out.`;
   287→      const res = await fetch("/api/ai/announce", {
   288→        method: "POST",
   289→        headers: { "Content-Type": "application/json" },
   290→        body: JSON.stringify({ text }),
   291→      });
   292→      if (!res.ok) throw new Error("tower silent");
   293→      const blob = await res.blob();
   294→      if (audioRef.current) URL.revokeObjectURL(audioRef.current.src);
   295→      const url = URL.createObjectURL(blob);
   296→      audioRef.current = new Audio(url);
   297→      audioRef.current.play();
   298→    } catch {
   299→      toast({
   300→        title: planMode === "road" ? "The conductor is not answering" : "Tower is not answering",
   301→        description: "The announcement will have to be imagined.",
   302→      });
   303→    } finally {
   304→      setAnnouncing(false);
   305→    }
   306→  }

   332→      setPostcardLoading(false);
