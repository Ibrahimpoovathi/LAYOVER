   324→  async function makePostcard() {
   325→    if (!plan) return;
   326→    setPostcardLoading(true);
   327→    setPostcardUrl(null);
   328→    setPostcardFallback(false);
   329→    try {
   330→      // Postcards are sent from where you're headed: in WHEELS the stop is the
   331→      // origin, so illustrate the destination town instead.
   332→      const postcardCity =
   333→        planMode === "road" ? (plan.destination?.city ?? plan.airport.city) : plan.airport.city;
   334→      const res = await fetch("/api/ai/postcard", {
   335→        method: "POST",
   336→        headers: { "Content-Type": "application/json" },
   337→        body: JSON.stringify({
   338→          city: postcardCity,
   339→          country: plan.airport.country ?? "",
   340→          vibe: plan.vibe,
   341→          mode: planMode,
   342→          vehicle: plan.vehicle,
   343→        }),
   344→      });
   345→      const data = await res.json();
   346→      if (!data.ok || !data.id) {
   347→        setPostcardFallback(true);
   348→        return;
   349→      }
   350→      // Poll the press with short requests — no proxy can sever those.
   351→      const deadline = Date.now() + 150_000;
   352→      let polls = 0;
   353→      let pollErrors = 0;
   354→      while (Date.now() < deadline) {
   355→        await new Promise((r) => setTimeout(r, polls === 0 ? 1_500 : 2_500));
   356→        polls++;
   357→        try {
   358→          const pr = await fetch(`/api/ai/postcard?id=${encodeURIComponent(data.id)}`);
   359→          if (!pr.ok) throw new Error(`poll ${pr.status}`);
   360→          const pj = await pr.json();
   361→          if (pj.status === "done" && pj.imageDataUrl) {
   362→            setPostcardUrl(pj.imageDataUrl);
   363→            return;
   364→          }
   365→          if (pj.status === "jammed") {
   366→            setPostcardFallback(true);
   367→            return;
   368→          }
   369→        } catch {
   370→          pollErrors++;
   371→          if (pollErrors >= 4) {
   372→            setPostcardFallback(true);
   373→            return;
   374→          }
   375→        }
   376→      }
   377→      setPostcardFallback(true); // press overran its deadline
   378→    } catch {
   379→      setPostcardFallback(true);
   380→    } finally {
   381→      setPostcardLoading(false);
   382→    }
   383→  }
